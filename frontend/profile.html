async function loadUserLibrary() { const token = localStorage.getItem('token'); const res = await fetch('/api/books/library', { headers: { 'Authorization': `Bearer ${token}` } }); const books = await res.json(); const grid = document.getElementById('my-library-grid'); grid.innerHTML = ''; if (books.length === 0) { grid.innerHTML = `

Поки що порожньо.

`; return; } books.reverse().forEach(book => { const card = document.createElement('div'); card.className = 'mini-book-card'; card.style.cursor = 'pointer'; const coverHtml = book.cover_filepath ? `` : `
📖
`; card.innerHTML = ` В бібліотеці ${coverHtml}
${book.title}
`; card.onclick = () => { // Navigate to home and open book window.location.href = `/#book-${book.id}`; }; grid.appendChild(card); }); }
